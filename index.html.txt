  let cart = {};

  function renderProducts(filter='all'){
    const grid = document.getElementById('productGrid');
    grid.innerHTML = '';
    products.filter(p => filter==='all' || p.platform===filter).forEach(p => {
      const card = document.createElement('div');
      card.className = 'card';
      card.innerHTML = `
        ${(p.id==='ig-takip-1000' || p.id==='tt-begeni-1000') ? '<span class="pop-badge">Popüler</span>' : ''}
        <div class="card-icon ${p.iconClass}">${p.icon}</div>
        <div class="card-platform">${p.label}</div>
        <div class="card-title">${p.title}</div>
        <div class="card-desc">${p.desc}</div>
        <div class="card-price-row">
          <div class="card-price">${p.price}<sup>TL</sup></div>
          ${p.est ? '<span class="est-badge">TAHMİNİ</span>' : ''}
        </div>
        <button class="add-btn" onclick="addToCart('${p.id}')">Sepete Ekle</button>
      `;
      grid.appendChild(card);
    });
  }

  function addToCart(id){
    cart[id] = (cart[id] || 0) + 1;
    updateCartUI();
    showToast('Sepete eklendi ✓');
  }
  function changeQty(id, delta){
    cart[id] = (cart[id] || 0) + delta;
    if(cart[id] <= 0) delete cart[id];
    updateCartUI();
  }
  function removeItem(id){
    delete cart[id];
    updateCartUI();
  }

  function updateCartUI(){
    const count = Object.values(cart).reduce((a,b)=>a+b,0);
    document.getElementById('cartCount').textContent = count;

    const itemsEl = document.getElementById('drawerItems');
    itemsEl.innerHTML = '';
    let total = 0;

    if(count === 0){
      itemsEl.innerHTML = '<div class="empty-cart">Sepetin şu an boş.<br>Ürün ekleyerek başla ❄</div>';
    } else {
      Object.entries(cart).forEach(([id, qty]) => {
        const p = products.find(x => x.id === id);
        const lineTotal = p.price * qty;
        total += lineTotal;
        const row = document.createElement('div');
        row.className = 'drawer-item';
        row.innerHTML = `
          <div>
            <div class="di-title">${p.title}</div>
            <div class="di-price">${p.price} TL x ${qty} = ${lineTotal} TL</div>
          </div>
          <div class="qty-ctrl">
            <button onclick="changeQty('${id}', -1)">-</button>
            <span>${qty}</span>
            <button onclick="changeQty('${id}', 1)">+</button>
            <button class="remove-btn" onclick="removeItem('${id}')">Sil</button>
          </div>
        `;
        itemsEl.appendChild(row);
      });
    }
    document.getElementById('totalPrice').textContent = total + ' TL';
    document.getElementById('checkoutBtn').disabled = count === 0;
  }

  function openCart(){
    document.getElementById('drawer').classList.add('show');
    document.getElementById('overlay').classList.add('show');
  }
  function closeCart(){
    document.getElementById('drawer').classList.remove('show');
    document.getElementById('overlay').classList.remove('show');
  }

  function showToast(msg){
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(()=> t.classList.remove('show'), 1800);
  }

  function checkout(){
    const lines = Object.entries(cart).map(([id, qty]) => {
      const p = products.find(x => x.id === id);
      return `${p.title} x${qty} = ${p.price*qty}TL`;
    });
    const total = Object.entries(cart).reduce((sum,[id,qty]) => {
      const p = products.find(x => x.id === id);
      return sum + p.price*qty;
    }, 0);
    const message = encodeURIComponent(`Merhaba, Frost Medya üzerinden sipariş vermek istiyorum:\n${lines.join('\n')}\nToplam: ${total} TL`);
    window.open(`https://wa.me/905316618975?text=${message}`, '_blank');
  }

  function initParticles(){
    const container = document.getElementById('particles');
    const count = window.innerWidth < 640 ? 14 : 26;
    for(let i=0; i<count; i++){
      const el = document.createElement('div');
      el.className = 'flake';
      const size = Math.random()*3 + 1.5;
      el.style.width = size+'px';
      el.style.height = size+'px';
      el.style.left = Math.random()*100 + 'vw';
      el.style.animationDuration = (Math.random()*18 + 14) + 's';
      el.style.animationDelay = (Math.random()*-20) + 's';
      container.appendChild(el);
    }
  }
